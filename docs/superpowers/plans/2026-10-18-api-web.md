# API pour le web — plan d'implémentation

> **Pour les agents :** sous-skill requise : superpowers:executing-plans (exécution inline).
> Les étapes se cochent (`- [ ]`).

**Objectif :** donner à l'API ce que le nouveau web lit et écrit : dossiers et notes de routine,
la dernière séance d'une routine, les séances d'une routine, les exercices faits.

**Architecture :** on suit le motif existant — schémas Zod et types dans `shared`, fonctions
pures testées dans `services/`, un dépôt Prisma injecté, des routes minces derrière
`exigerSession`, testées avec un dépôt en mémoire et un lecteur de session factice.

**Pile :** Prisma 7 + PostgreSQL (Neon), Hono 4, Zod 4, Vitest.

**Spec :** `docs/superpowers/specs/2026-10-18-web-api-design.md` (D296–D298, D310–D312).

## Contraintes globales

- Branche `feat/api-web` depuis `origin/main`, dans un worktree `../better-lift-api-web` (même
  raison que l'audit). Si l'audit est déjà fusionné, partir de lui.
- Migration sur la branche Neon **`dev`**, base **`betterlift`**, par la skill `/migration`.
  Jamais en production sans « vas-y » explicite ; la prod migre au build Vercel.
- Toute route de données personnelles passe par `exigerSession`. La donnée d'autrui : 404.
- Toute erreur : `{ erreur }`. Chaque `c.json` porte son code (D295).
- Le mobile ne perd rien : `note` et `dossierId` **absents** d'une saisie = inchangés (D298).
- `shared` : TypeScript pur, pas de Prisma, pas de `crypto.randomUUID`.
- Commits Conventional Commits en français, sujet en minuscule, ≤ 100 caractères.

```ts
async function verifierDossier(
```

```ts
import type { Dossier } from "@better-lift/shared";
import { Hono } from "hono";
import { describe, expect, it } from "vitest";
import type { LireSession } from "../middlewares/session.js";
import { type DepotDossiers, NomPris } from "../services/dossiers.js";
import { routesDossiers } from "./dossiers.js";

function fauxDepot(): DepotDossiers {
  const dossiers = new Map<string, Dossier & { userId: string }>();
  let n = 1;
  const pris = (userId: string, nom: string, sauf?: string) =>
    [...dossiers.values()].some((d) => d.userId === userId && d.nom === nom && d.id !== sauf);
  return {
    async lister(userId) {
      return [...dossiers.values()].filter((d) => d.userId === userId).map(({ id, nom }) => ({ id, nom }));
    },
    async creer(userId, nom) {
      if (pris(userId, nom)) throw new NomPris();
      const id = `d${n++}`;
      dossiers.set(id, { id, nom, userId });
      return id;
    },
    async renommer(userId, id, nom) {
      const d = dossiers.get(id);
      if (!d || d.userId !== userId) return false;
      if (pris(userId, nom, id)) throw new NomPris();
      d.nom = nom;
      return true;
    },
    async supprimer(userId, id) {
      const d = dossiers.get(id);
      return !!d && d.userId === userId && dossiers.delete(id);
    },
    async ordonner(userId, ids) {
      return ids.every((id) => dossiers.get(id)?.userId === userId);
    },
  };
}

const lireSession: LireSession = async (h) => {
```

```ts
  typeof e === "object" && e !== null && "code" in e && e.code === "P2002";

export const depotDossiers: DepotDossiers = {
  async lister(userId) {
    return prisma.dossier.findMany({
      where: { userId },
      orderBy: [{ position: "asc" }, { createdAt: "asc" }],
      select: { id: true, nom: true },
    });
  },
  async creer(userId, nom) {
    const dernier = await prisma.dossier.aggregate({ where: { userId }, _max: { position: true } });
    try {
      const d = await prisma.dossier.create({
        data: { userId, nom, position: (dernier._max.position ?? -1) + 1 },
        select: { id: true },
      });
      return d.id;
    } catch (e) {
      if (estNomPris(e)) throw new NomPris();
      throw e;
    }
  },
  async renommer(userId, id, nom) {
    try {
      const { count } = await prisma.dossier.updateMany({ where: { id, userId }, data: { nom } });
      return count > 0;
    } catch (e) {
      if (estNomPris(e)) throw new NomPris();
      throw e;
    }
  },
  async supprimer(userId, id) {
    const { count } = await prisma.dossier.deleteMany({ where: { id, userId } });
    return count > 0;
  },
  async ordonner(userId, ids) {
    return prisma.$transaction(async (tx) => {
      const siens = await tx.dossier.count({ where: { userId, id: { in: ids } } });
      if (siens !== ids.length) return false;
      await Promise.all(ids.map((id, position) => tx.dossier.update({ where: { id }, data: { position } })));
      return true;
    });
  },
};
```

```ts
import { recordsDe, type SerieHisto } from "./records.js";

/** Ce que la page Exercices joint au catalogue, pour un exercice fait (D312). */
export type ExerciceFait = {
  slug: string;
  /** ISO de la dernière séance où il figure. */
  derniereFois: string;
  /** Le meilleur 1RM estimé (record `UN_RM`), `null` sans série chargée qui compte. */
  meilleur1Rm: number | null;
};

/**
 * Par exercice, la dernière fois et le meilleur 1RM. Le 1RM vient de `recordsDe` : mêmes règles
 * que les records (séries faites, hors échauffement, 12 reps au plus).
 */
export function resumerExercicesFaits(
  series: readonly (SerieHisto & { slug: string; date: string })[],
): ExerciceFait[] {
  const parSlug = new Map<string, (SerieHisto & { date: string })[]>();
  for (const { slug, ...serie } of series) {
    const liste = parSlug.get(slug) ?? [];
    liste.push(serie);
    parSlug.set(slug, liste);
  }
  return [...parSlug].map(([slug, liste]) => {
    const chrono = [...liste].sort((a, b) => Date.parse(a.date) - Date.parse(b.date));
    const unRm = recordsDe(chrono).find((r) => r.type === "UN_RM");
    return {
      slug,
      derniereFois: chrono.at(-1)?.date ?? "",
      meilleur1Rm: unRm?.valeur ?? null,
    };
  });
}
```
