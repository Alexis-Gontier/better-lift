# Brancher le web sur l'API — plan d'implémentation

> **Pour les agents :** sous-skill requise : superpowers:executing-plans (exécution inline,
> voulue par l'utilisateur). Les étapes se cochent (`- [ ]`). Un lot = un ou plusieurs commits,
> des captures, une validation de l'utilisateur avant le lot suivant.

**Objectif :** le web (wireframe de `feat/web-wireframe`) lit et écrit l'API réelle ; `mocks/`
disparaît ; une suite Playwright prouve chaque page de bout en bout.

**Architecture :** `lib/` porte la config, le client RPC, l'erreur d'API, l'auth et la locale ;
chaque feature a son `queries.ts` (clés, `queryOptions`, mutations) ; les routes préchargent
sans attendre et les pages choisissent leur état par `etatAffiche`. Les écrans consomment les
types de l'API ; les calculs viennent de `shared`.

**Pile :** Vite + React 19, TanStack Router 1.170 + TanStack Query 5.103, Better Auth 1.7.5
(client React), Hono RPC (`@better-lift/api-client`), Vitest, Playwright.

**Spec :** `docs/superpowers/specs/2026-10-18-web-api-design.md` (D293–D313).

## Contraintes globales

- Prérequis : plans `api-audit` et `api-web` fusionnés dans `main`. Rebaser
  `feat/web-wireframe` sur `main`, puis créer `feat/web-api` depuis elle.
- Les fichiers non commités de l'utilisateur (`register-form.tsx`, spec records) : ne pas les
  embarquer ; lui demander quoi en faire avant le rebase.
- Versions exactes, comme sur `main` : `@tanstack/react-query` **5.103.2**, `better-auth`
  **1.7.5**, `zod` **4.6.5**. `@playwright/test` : la dernière, épinglée exacte.
- L'application vit sous `/app` (D292) : jamais de `/app` en dur dans le code, le routeur porte
  `basepath`. L'API est sous `/api` en production, sur le 3001 en développement.
- Fuseau : celui du navigateur ; unités : métriques ; les deux fixés dans `lib/locale.ts` (D313).
  Tout découpage de jour passe par le calendrier de `shared`, jamais par l'UTC.
- Une feature n'importe pas une autre feature (D306). `lib/` sans React.
- Clés de cache : toute donnée de la clé de lecture est dans la clé invalidée à l'écriture.
- Formulaires : tester l'état de chargement avant `!data`.
- E2E : compte `e2e@better-lift.local` seulement, jamais `controle@` ; données préfixées `e2e-`.
- Capture Playwright de chaque page dans ses quatre états après chaque lot
  ([[verifier-apres-changement]]).
- Commits Conventional Commits en français, sujet en minuscule, ≤ 100 caractères.

`apps/web/src/lib/api.ts` :

```ts
import { creerClient } from "@better-lift/api-client";
import { URL_API } from "./config";

/** L'URL de base est la seule chose propre au web : le client, lui, est partagé. */
export const api = creerClient(URL_API);

/** Un échec d'API, avec le message de `{ erreur }` et le statut ; 0 : serveur injoignable. */
export class ErreurApi extends Error {
  constructor(
    message: string,
    readonly statut: number,
  ) {
    super(message);
    this.name = "ErreurApi";
  }
}

const INJOIGNABLE = "Le serveur est injoignable. Vérifiez votre connexion, puis réessayez.";

type Reponse = { ok: boolean; status: number; json(): Promise<unknown> };

/**
 * Lance l'appel et rend la réponse si elle est `ok`, rétrécie au type du succès ; lève une
 * `ErreurApi` sinon. Générique pour garder le typage RPC : un `Response` nu ferait de
 * `res.json()` un `any`.
 */
export async function lire<R extends Reponse>(
  appel: () => Promise<R>,
  parDefaut: string,
): Promise<Extract<R, { ok: true }>> {
  let res: R;
  try {
    res = await appel();
  } catch {
    throw new ErreurApi(INJOIGNABLE, 0);
  }
  if (!res.ok) {
    const corps = (await res.json().catch(() => null)) as { erreur?: unknown } | null;
    throw new ErreurApi(typeof corps?.erreur === "string" ? corps.erreur : parDefaut, res.status);
  }
  return res as Extract<R, { ok: true }>;
}
```

```ts
export const ETATS = ["nominal", "vide", "chargement", "erreur"] as const;
export type Etat = (typeof ETATS)[number];

const estEtat = (v: unknown): v is Etat => typeof v === "string" && (ETATS as readonly string[]).includes(v);

/** La recherche commune à toutes les pages : l'état forcé (revue) et l'onglet. */
export function parseEtat(search: Record<string, unknown>): { etat?: Etat; onglet?: string } {
  const out: { etat?: Etat; onglet?: string } = {};
  if (estEtat(search.etat)) out.etat = search.etat;
  if (typeof search.onglet === "string" && search.onglet !== "") out.onglet = search.onglet;
  return out;
}

/**
 * L'état qu'une page montre (D303) : celui que force l'URL (`?etat=`, pour la revue), sinon
 * celui de sa requête.
 */
export function etatAffiche(
  force: Etat | undefined,
  requete: { isPending: boolean; isError: boolean },
  vide: boolean,
): Etat {
  if (force) return force;
  if (requete.isPending) return "chargement";
  if (requete.isError) return "erreur";
  return vide ? "vide" : "nominal";
}
```

```ts
import { createAuthClient } from "better-auth/react";
import { URL_AUTH } from "./config";

/** `credentials: "include"` : en développement, le web (3000) et l'API (3001) diffèrent. */
export const authClient = createAuthClient({
  baseURL: URL_AUTH,
  fetchOptions: { credentials: "include" },
});

export type Session = typeof authClient.$Infer.Session | null;

/**
 * L'objet passé au contexte du routeur. Il est MUTÉ puis le routeur invalidé, ce qui fait
 * relire `beforeLoad` sans recréer le routeur.
 */
export const auth: { session: Session } = { session: null };

/**
 * Lue AVANT le premier rendu. Une API injoignable vaut « pas de session » : on retombe sur la
 * connexion, qui dira l'échec à la tentative — jamais une page blanche (D301).
 */
export async function chargerSession(): Promise<void> {
  try {
    const { data } = await authClient.getSession();
    auth.session = data ?? null;
  } catch {
    auth.session = null;
  }
}

/** Où revenir après la connexion : un chemin interne seulement, sinon l'Accueil. */
export function cibleApresConnexion(redirect: unknown): string {
  return typeof redirect === "string" && redirect.startsWith("/") && !redirect.startsWith("//")
    ? redirect
    : "/";
}
```

- [ ] **Étape 2 : implémentation**

```ts
/**
 * Les suppressions différées (D302) : la ligne quitte l'écran tout de suite, l'appel ne part
 * qu'à l'échéance. « Annuler » n'a donc rien à défaire côté serveur. Sans React : le hook vit
 * dans `components/use-suppression.ts`.
 */
export function creerDifferes(delaiMs: number) {
  const enAttente = new Map<string, { minuteur: ReturnType<typeof setTimeout>; executer: () => Promise<void> }>();
  return {
    programmer(cle: string, executer: () => Promise<void>) {
      const minuteur = setTimeout(() => {
        enAttente.delete(cle);
        void executer();
      }, delaiMs);
      enAttente.set(cle, { minuteur, executer });
    },
    annuler(cle: string) {
      const e = enAttente.get(cle);
      if (!e) return false;
```

```ts
import { type QueryKey, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";
import { differes } from "@/lib/differe";

/**
 * Supprimer avec « Annuler » (D302). `retirer` enlève l'élément des données en cache ;
 * `supprimer` fait l'appel (avec `{ init: { keepalive: true, credentials: "include" } }`).
 */
export function useSuppression<T>(cle: QueryKey, options: {
  retirer: (donnees: T, id: string) => T;
  supprimer: (id: string) => Promise<void>;
  libelle: string;
}) {
  const client = useQueryClient();
  return (id: string) => {
    const avant = client.getQueryData<T>(cle);
    if (avant !== undefined) client.setQueryData<T>(cle, options.retirer(avant, id));
    const restaurer = () => avant !== undefined && client.setQueryData(cle, avant);
    differes.programmer(id, async () => {
      try {
        await options.supprimer(id);
      } catch (e) {
        restaurer();
        toast.error(e instanceof Error ? e.message : "La suppression a échoué.");
      } finally {
        await client.invalidateQueries({ queryKey: cle });
      }
    });
    toast(options.libelle, {
      action: { label: "Annuler", onClick: () => differes.annuler(id) && restaurer() },
    });
  };
}
```

```ts
import type { RoutineDetail, SeanceResume } from "@better-lift/shared";

/** Les stats d'une routine sur ses séances (D280, D311), la plus ancienne d'abord. */
export function statsRoutine(routine: RoutineDetail, seances: readonly SeanceResume[], depuis: string) {
  const runs = seances
    .filter((s) => s.debut >= depuis)
    .sort((a, b) => a.debut.localeCompare(b.debut));
  if (runs.length === 0) return null;
  const prevues = routine.exercices.reduce(
    (n, e) => n + e.series.filter((s) => s.type !== "ECHAUFFEMENT").length, 0);
  const moyenne = (f: (s: SeanceResume) => number) => runs.reduce((n, s) => n + f(s), 0) / runs.length;
  return {
    nb: runs.length,
    volumeMoyen: moyenne((s) => s.volumeKg),
    dureeMoyenneMin: moyenne((s) => (Date.parse(s.fin) - Date.parse(s.debut)) / 60_000),
    records: runs.reduce((n, s) => n + s.nbRecords, 0),
    realisation: prevues > 0 ? moyenne((s) => Math.min(1, s.seriesFaites / prevues)) : 1,
    volumes: runs.map((s) => ({ date: s.debut, value: s.volumeKg })),
  };
}
```
